const dow = new Date().getDay();
const todayName = DAYS[dow-1] || null;
let selectedDay = todayName || "Monday";

function toMinutes(hhmm){
  const [h,m] = hhmm.split(":").map(Number);
  return h*60+m;
}
function nowMinutes(){
  const d = new Date();
  return d.getHours()*60 + d.getMinutes();
}
function fmtMinLeft(mins){
  if (mins < 60) return {num: mins, label: mins===1 ? "minute left" : "minutes left"};
  const h = Math.floor(mins/60), m = mins%60;
  return {num: h + "h" + (m? " "+m+"m":""), label: "left"};
}

const scheduleEl = document.getElementById('schedule');
const searchInput = document.getElementById('searchInput');

function matchesSearch(item, q){
  if (!q) return true;
  const hay = [item.subject, item.room, item.teacher, item.code].join(' ').toLowerCase();
  return hay.includes(q);
}

function renderSchedule(){
  const q = searchInput.value.trim().toLowerCase();
  const items = timetable[selectedDay];
  scheduleEl.innerHTML = '';

  if (!items.length){
    scheduleEl.innerHTML = '<div class="empty-day">No classes scheduled.</div>';
    return;
  }

  const isToday = selectedDay === todayName;
  const nm = nowMinutes();

  let anyVisible = false;
  items.forEach((item)=>{
    if (item.isBreak){
      if (q) return;
      const row = document.createElement('div');
      row.className = 'break-slot';
      row.innerHTML = `<div class="slot-time">${item.start}</div><div>${item.label}</div>`;
      scheduleEl.appendChild(row);
      return;
    }
    if (!matchesSearch(item, q)) return;
    anyVisible = true;

    const s = toMinutes(item.start), e = toMinutes(item.end);
    const isNow = isToday && nm >= s && nm < e && !q;
    const isPast = isToday && nm >= e;

    const row = document.createElement('div');
    row.className = 'slot cat-' + item.cat + (isNow ? ' now-slot':'') + (isPast && !q ? ' dimmed':'');
    row.innerHTML = `
      <div class="slot-time utility">${item.start}<br>${item.end}</div>
      <div class="slot-bar"></div>
      <div class="slot-body">
        ${isNow ? '<span class="now-pill">NOW</span><br>' : ''}
        <div class="slot-subject">${item.subject}</div>
        <div class="slot-meta">${item.room} · ${item.teacher}</div>
      </div>
    `;
    row.addEventListener('click', ()=> openSheet(item));
    scheduleEl.appendChild(row);
  });

  if (q && !anyVisible){
    scheduleEl.innerHTML = '<div class="empty-day">No matches for “' + searchInput.value + '”.</div>';
  }
}

function findCurrentAndNext(){
  if (!todayName) return {current:null, next:null};
  const items = timetable[todayName];
  const nm = nowMinutes();
  let current = null, next = null;
  for (const item of items){
    const s = toMinutes(item.start), e = toMinutes(item.end);
    if (nm >= s && nm < e){ current = item; }
    if (nm < s && !next){ next = item; }
  }
  return {current, next};
}

function renderNowCard(){
  const label = document.getElementById('nowLabel');
  const subject = document.getElementById('nowSubject');
  const detail = document.getElementById('nowDetail');
  const timerNum = document.getElementById('nowTimerNum');
  const timerLabel = document.getElementById('nowTimerLabel');
  const progress = document.getElementById('nowProgress');

  if (!todayName){
    label.textContent = "It's the weekend";
    subject.textContent = "No school today";
    detail.textContent = "Back on Monday";
    timerNum.textContent = "—";
    timerLabel.textContent = "";
    progress.style.width = '0%';
    return;
  }

  const { current, next } = findCurrentAndNext();
  const nm = nowMinutes();

  if (current){
    const s = toMinutes(current.start), e = toMinutes(current.end);
    const left = e - nm;
    const total = e - s;
    label.textContent = 'In progress';
    subject.textContent = current.subject;
    detail.textContent = current.room + ' · ' + current.teacher;
    const f = fmtMinLeft(left);
    timerNum.textContent = f.num;
    timerLabel.textContent = f.label;
    progress.style.width = Math.min(100, Math.max(0, ((nm-s)/total)*100)) + '%';
  } else if (next){
    const s = toMinutes(next.start);
    const until = s - nm;
    label.textContent = 'Up next';
    subject.textContent = next.subject;
    detail.textContent = next.room + ' · ' + next.teacher + ' · starts ' + next.start;
    const f = fmtMinLeft(until);
    timerNum.textContent = f.num;
    timerLabel.textContent = f.label==='left' ? 'until start' : 'until start';
    progress.style.width = '0%';
  } else {
    label.textContent = 'Done for today';
    subject.textContent = 'School day finished';
    detail.textContent = 'See you tomorrow';
    timerNum.textContent = '🎉';
    timerLabel.textContent = '';
    progress.style.width = '100%';
  }
}
